Script

$ curl -sSL https://localai.io/install.sh | sh
$ local-ai run qwen3-4b
# the API and the web UI are on http://localhost:8080

Docker

$ docker run -p 8080:8080 --name local-ai -ti localai/localai:latest
# Podman works the same way

Kubernetes

$ helm repo add go-skynet https://go-skynet.github.io/helm-charts/
$ helm install local-ai go-skynet/local-ai
# or: kubectl apply -f https://localai.io/install/kubernetes.yaml
